        Optional: iLO Redfish proxy helper (Node.js)
        
For many iLOs, browsers are blocked by CORS. Run this tiny proxy locally then use the wizard's discovery button.

        
// save as redfish-proxy.js
import http from 'node:http';
import https from 'node:https';
import { URL } from 'node:url';
const PORT = 8787;
http.createServer((req,res)=>{
  const u = new URL(req.url, `http://localhost:${PORT}`);
  const target = u.searchParams.get('url');
  const user = u.searchParams.get('user')||'';
  const pass = u.searchParams.get('pass')||'';
  if(!target){ res.statusCode=400; return res.end('Missing url'); }
  const t = new URL(target);
  const opts = { hostname:t.hostname, port:t.port||443, path:t.pathname+t.search, method:'GET', rejectUnauthorized:false, headers:{ 'Authorization':'Basic '+Buffer.from(`${user}:${pass}`).toString('base64') } };
  https.request(opts, r=>{ res.setHeader('Access-Control-Allow-Origin','*'); r.pipe(res); }).on('error', e=>{ res.statusCode=502; res.end(String(e)); }).end();
}).listen(PORT, ()=> console.log('proxy on :'+PORT));
